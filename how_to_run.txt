# DPI-LS Observability Stack – Complete Run Guide
# (Real Prometheus 3.12 + Real Grafana 13 + Phoenix + MLflow + Backend)
# =====================================================================
# Open 4 separate PowerShell terminals. Run each block in its terminal.
# Wait for each to start before going to the next.
# =====================================================================


# ══════════════════════════════════════════════════════════════════
# TERMINAL 1 – DPI-LS Backend (FastAPI, port 8000)
# This also exposes /metrics for Prometheus to scrape
# ══════════════════════════════════════════════════════════════════
Set-ExecutionPolicy -Scope Process -ExecutionPolicy RemoteSigned
& d:\Projects\widget\widget\.venv\Scripts\Activate.ps1
cd d:\Projects\widget\widget
uv run uvicorn api.app:app --reload --host 127.0.0.1 --port 8000

# Wait until you see: INFO: Application startup complete.


# ══════════════════════════════════════════════════════════════════
# TERMINAL 2 – Arize Phoenix (LLM Observability, port 6006)
# ══════════════════════════════════════════════════════════════════
Set-ExecutionPolicy -Scope Process -ExecutionPolicy RemoteSigned
& d:\Projects\widget\widget\.venv\Scripts\Activate.ps1
cd d:\Projects\widget\widget
$env:DPI_LS_NO_BLOCK="1"
uv run phoenix serve

# Wait until you see: Phoenix UI available at: http://localhost:6006


# ══════════════════════════════════════════════════════════════════
# TERMINAL 3 – MLflow (port 5000)
# ══════════════════════════════════════════════════════════════════
Set-ExecutionPolicy -Scope Process -ExecutionPolicy RemoteSigned
& d:\Projects\widget\widget\.venv\Scripts\Activate.ps1
cd d:\Projects\widget\widget
uv run mlflow server --host 127.0.0.1 --port 5000

# Wait until you see: INFO: Uvicorn running on http://127.0.0.1:5000


# ══════════════════════════════════════════════════════════════════
# TERMINAL 4 – Real Prometheus 3.12 (port 9090)
# Scrapes backend /metrics, Phoenix, and MLflow
# ══════════════════════════════════════════════════════════════════
cd "C:\Users\User\Downloads\prometheus-3.12.0.windows-amd64\prometheus-3.12.0.windows-amd64"
.\prometheus.exe --config.file=prometheus.yml --web.listen-address=127.0.0.1:9090

# Wait until you see: Server is ready to receive web requests.


# ══════════════════════════════════════════════════════════════════
# TERMINAL 5 – Real Grafana 13 (port 3000)
# Auto-loads all 6 data sources and DPI-LS dashboard
# ══════════════════════════════════════════════════════════════════
cd "C:\Users\User\Downloads\grafana_13.0.2_26816849631_windows_amd64\grafana-13.0.2\bin"
.\grafana.exe server --homepath ".."

# Wait until you see: HTTP Server Listen on [::]:3000
# Then open: http://localhost:3000
# Login: admin / admin  (change password: skip or set to "admin")


# ══════════════════════════════════════════════════════════════════
# OPTIONAL – Send a test score to backend (generates live metrics)
# Run in any terminal that has venv active
# ══════════════════════════════════════════════════════════════════
& d:\Projects\widget\widget\.venv\Scripts\Activate.ps1
cd d:\Projects\widget\widget
$env:DPI_LS_NO_BLOCK="1"
uv run python examples/test_agent.py
